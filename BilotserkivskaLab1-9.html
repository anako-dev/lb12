Програмування

#include <iostream>
#include <Windows.h>
using namespace std;

struct Node {
    int data;
    Node* next;
    Node* prev;
    Node(int value) : data(value), next(nullptr), prev(nullptr) {}
};

class Deque {
private:
    Node* right;
    Node* left;
    int size;
public:
    Deque() : right(nullptr), left(nullptr), size(0) {}
    void pushRight(int value) {
        Node* newNode = new Node(value);
        if (right == nullptr) {
            right = left = newNode;
        } else {
            newNode->next = right;
            right->prev = newNode;
            right = newNode;
        }
        size++;
    }
};
